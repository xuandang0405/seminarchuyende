// Mở ứng dụng và nạp nội dung đa ngôn ngữ
// H: mô hình hóa từ HTML, có cụ thể hóa thiết kế
// Use case: T01, T03, T05, T07, F07
// Đây là mô hình đề xuất, không phải sơ đồ trích xuất từ backend đã chạy.
// API là ranh giới FastAPI; chi tiết Router, Service, Repository được tách ở sơ đồ backend.
// Cache gồm IndexedDB cho snapshot và Cache API cho tài nguyên theo hướng PWA của HTML.
// ETag chỉ được gửi khi client còn snapshot tương ứng với đúng ngôn ngữ. Các quyết định về request token và merge được đề xuất để tránh dữ liệu cũ ghi đè.

autoNumber on
colorMode pastel
styleMode plain
typeface clean

User [label: "Du khách", color: blue]
App [label: "Ứng dụng", color: blue]
Cache [label: "Bộ nhớ offline", color: green]
API [label: "FastAPI", color: orange]

User > App: Mở ứng dụng và chọn ngôn ngữ
App > App: Ghi nhận ngôn ngữ đích và request token
App > Cache: Đọc snapshot POI và UI bundle đã lưu
Cache --> App: Trả dữ liệu đã có hoặc rỗng
opt [label: "Có snapshot khả dụng"] {
  App --> User: Hiển thị dữ liệu đã lưu trong lúc kiểm tra cập nhật
}
alt [label: "Có mạng"] {
  par [label: "Nạp nội dung POI"] {
    App > API: Yêu cầu POI theo ngôn ngữ và ETag hợp lệ
    alt [label: "Dữ liệu không đổi"] {
      API --> App: 304 Not Modified
    }
    else [label: "Trường hợp còn lại"] {
      alt [label: "Có dữ liệu mới"] {
        API --> App: Snapshot hoặc delta và danh sách POI đã gỡ
        App > Cache: Merge theo chế độ full hoặc delta và lưu version
      }
      else [label: "Lỗi hoặc timeout"] {
        API --> App: Báo không tải được dữ liệu
        App > App: Giữ snapshot đang có
      }
    }
  }
  and [label: "Nạp bản dịch giao diện"] {
    App > API: Yêu cầu UI bundle cho ngôn ngữ đích
    API --> App: Bundle ready, English fallback pending hoặc lỗi
    alt [label: "Có bundle dùng được"] {
      App > Cache: Lưu bundle kèm locale và source hash
    }
    else [label: "Không tải được"] {
      App > App: Giữ bundle cục bộ hoặc giao diện mặc định
    }
  }
  App > App: Bỏ kết quả cũ nếu request token đã thay đổi
  alt [label: "Có nội dung và giao diện khả dụng"] {
    App --> User: Hiển thị cùng ngôn ngữ phù hợp hoặc fallback đã chọn
  }
  else [label: "Chưa có đủ nội dung sử dụng"] {
    App --> User: Thông báo lỗi tải và cho phép thử lại
  }
}
else [label: "Không có mạng"] {
  alt [label: "Có dữ liệu đã tải"] {
    App --> User: Mở chế độ offline với ngôn ngữ khả dụng
  }
  else [label: "Chưa tải nội dung"] {
    App --> User: Thông báo cần mạng để tải dữ liệu lần đầu
  }
}

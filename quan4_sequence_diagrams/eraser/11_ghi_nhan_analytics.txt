// Dùng event ID ổn định cho mỗi sự kiện. Client chỉ bỏ các sự kiện server xác nhận hoặc từ chối vĩnh viễn, không xóa cả batch khi mới gửi.
// Đề xuất upsert event và đánh dấu kỳ cần tổng hợp trong cùng transaction; tránh insert trùng gây lỗi transaction rồi bỏ sót cập nhật.
// Rate limit có thể dùng Redis trong Service; chi tiết Redis được lược khỏi sơ đồ.
// Các thông số thời gian nghe nằm trong payload sự kiện; phải kiểm tra và không cộng lặp các giá trị tích lũy khi retry.

alt [label: "Người dùng chưa đồng ý"] {
  App > App: Không ghi thêm sự kiện public analytics
}
else [label: "Người dùng đã đồng ý"] {
  App > App: Ghi sự kiện vào outbox với event ID ổn định
  alt [label: "Chưa có mạng"] {
    App > App: Giữ outbox để gửi sau
  }
  else [label: "Có mạng"] {
    App > Router: Gửi batch sự kiện cùng ngữ cảnh đồng ý
    Router > Service: Chuyển batch đã kiểm tra định dạng
    Service > Service: Kiểm tra đồng ý, rate limit và payload
    alt [label: "Bị giới hạn tần suất"] {
      Service --> Router: Từ chối tạm thời với Retry-After
      Router --> App: 429 và thời gian chờ
      App > App: Giữ nguyên ID và thử lại sau
    }
    else [label: "Batch đủ điều kiện xử lý"] {
      Service > Repo: Ghi các event hợp lệ và đánh dấu kỳ cần tổng hợp
      Repo > DB: Transaction upsert theo event ID và cập nhật job tổng hợp
      alt [label: "Commit thành công"] {
        DB --> Repo: Các ID đã lưu hoặc đã tồn tại
        Repo --> Service: Danh sách kết quả cho từng sự kiện
        Service --> Router: ID đã nhận, đã có và bị từ chối
        Router --> App: 200 với kết quả chi tiết
        App > App: Bỏ mục đã nhận hoặc bị từ chối vĩnh viễn, giữ mục cần thử lại
      }
      else [label: "Mất kết nối hoặc chưa nhận được xác nhận"] {
        App > App: Giữ outbox và gửi lại cùng event ID
      }
    }
  }
}
